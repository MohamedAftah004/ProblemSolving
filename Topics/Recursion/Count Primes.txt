#include <iostream>
using namespace std;


bool IsPrime(int number , int testNumber = 3)
{
    if(number == 2)
        return true;
    
    if(number <= 1 || number % 2 == 0)
        return false;
    
    if(number == testNumber)
        return true;
    
    if(number % testNumber == 0)
        return false;
    
    return IsPrime(number , testNumber+1);
}

int CountPrimes(int start , int end)
{
    
    if(start >= end)
        return 0;
    
    int result = CountPrimes(start+1, end);
    if(IsPrime(start))
        result += 1;
    
    return result;
}


int main (){
    
    
    cout << CountPrimes(1 , 100) <<endl;
    
    
    return 0;
}
