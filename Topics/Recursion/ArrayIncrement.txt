#include <iostream>
using namespace std;

void ArrayIncremen(int arr[] ,  int arrLength )
{
    
    if(arrLength == 0)
        return ;

    ArrayIncremen(arr, arrLength-1);
    arr[arrLength-1] += arrLength-1;
    
}


int main (){
    
    int arr[10] = {1,2,3,4,5,23,11,8,9,10};
    ArrayIncremen(arr , 10);
    for(int i = 0 ; i<10 ; i++ )
    {
        cout << arr[i] << endl;
    }
    
    return 0;
}
