#include <iostream>
using namespace std;


int SuffixSum(int arr[] , int length , int count)
{
    if(count == 0)
        return 0;
    
    return arr[length-1] + SuffixSum(arr, length-1, count-1);
}

int PrefixSum_V2(int arr[] , int count)
{
    if(count == 0)
        return 0;
    
    return arr[0] + PrefixSum_V2(1+arr, count-1);
}



int main (){
    

    int arr[5] = {1 ,22, 33 ,44 ,55};
    int sum = SuffixSum(arr, 5 , 2);
    cout << sum << endl;
    
    return 0;
}
