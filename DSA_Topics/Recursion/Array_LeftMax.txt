#include <iostream>
using namespace std;



void ArrayLeftMax(int arr[] , int arrLength)
{
    
    if(arrLength == 1)
        return ;
   
    ArrayLeftMax(arr, arrLength-1);
    
    arr[arrLength-1] = max(arr[arrLength-1] , arr[arrLength-2]);
    
}



int main (){
    
    int arr[5] = {1,2,7,4,5};
    ArrayLeftMax(arr,5);
    for(int i = 0 ; i < 5 ; i++)
    {
        cout << arr[i] << endl;
    }
    
    return 0;
}
