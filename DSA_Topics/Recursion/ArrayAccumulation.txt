#include <iostream>
using namespace std;

void ArrayAccumulation(int arr[] , int arrLength )
{
    
    if(arrLength == 1)
        return ;
   

    ArrayAccumulation(arr, arrLength-1);
    
    arr[arrLength-1] += arr[arrLength-2];
    
}



int main (){
    
    int arr[5] = {1,2,3,4,5};
    ArrayAccumulation(arr,5); 
    for(int i = 0 ; i < 5 ; i++)
    {
        cout << arr[i] << endl;
    }
    
    return 0;
}
