#include <iostream>
using namespace std;

double ArrAverage(double arr[] , int length)
{
    if(length == 1 )
        return arr[0];
    
    double subResult = ArrAverage(arr, length-1);
    subResult = subResult*arr[length-1];
    return (subResult + arr[length-1])/length;
    
}


int main (){
    
    double arr[10] = {1,2,3,4,5,23,11,8,9,10};
    cout << ArrAverage(arr, 10) << endl ;
    
    return 0;
}
