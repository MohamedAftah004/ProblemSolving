#pragma once#include <iostream>#include "clsScreen.h";using namespace std;class clsListClientsScreen : protected clsScreen{#include "clsInputValidate.h"private:    static void _PrintClientRecordLine(clsBankClient Client)    {        cout << "\t| " << setw(15) << left << Client.AccountNumber();        cout << "| " << setw(20) << left << Client.FullName();        cout << "| " << setw(12) << left << Client.Phone;        cout << "| " << setw(20) << left << Client.Email;        cout << "| " << setw(10) << left << Client.PinCode;        cout << "| " << setw(12) << left << Client.AccountBalance;                       }        static void _ShowClientsList()    {        vector<clsBankClient>vClients = clsBankClient::GetClientsList();        cout << "\n\t__________________________________________________________________________________________________\n\n";        cout << "\t| " << left << setw(15) << "Account Number";        cout << "| " << left << setw(20) << "Client Name";        cout << "| " << left << setw(12) << "Phone";        cout << "| " << left << setw(20) << "Email";        cout << "| " << left << setw(10) << "PinCode";        cout << "| " << left << setw(12) << "Balance";        cout << "\n\t__________________________________________________________________________________________________\n\n";        for (clsBankClient C : vClients)        {            _PrintClientRecordLine(C);            cout << endl;        }        cout << "\n\t__________________________________________________________________________________________________\n";    }public:    static void ShowClientListScreen()    {        vector<clsBankClient>vClients = clsBankClient::GetClientsList();        short ClientsNumber = vClients.size();        string Title = "Client List Screen";        string SubTitle = "(" + to_string(ClientsNumber) + ") Client(s). ";        clsScreen::_DrawScreenHeader(Title, SubTitle);        _ShowClientsList();    }};
