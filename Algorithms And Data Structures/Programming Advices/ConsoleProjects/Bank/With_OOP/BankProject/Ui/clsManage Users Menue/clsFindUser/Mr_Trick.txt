	static void ShowFindUserScreen()	{		_DrawScreenHeader("Find User Screen");		string UserName = "";		cout << "\nPlease enter Username: ";		UserName = clsInputValidate::ReadString();		while (!clsUsers::IsUserExist(UserName))		{			cout << "\nUser with \"" << UserName << "\" is not found, choose another one: ";			UserName = clsInputValidate::ReadString();		}				clsUsers User = clsUsers::Find(UserName);		if (!User.IsEmpty())		{			cout << "\nUser with \"" << UserName << "\" was Founded :-) \n";		}		else		{			cout << "\nUser with \"" << UserName << "\" was not Found. \n";		}		_PrintUserCard(User);	}
