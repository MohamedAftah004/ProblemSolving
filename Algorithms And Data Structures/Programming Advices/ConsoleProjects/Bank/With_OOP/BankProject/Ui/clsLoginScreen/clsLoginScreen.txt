#pragma once#include <iostream>#include "clsUsers.h";#include "clsMainMenue.h";#include "Global.h";#include "clsScreen.h";using namespace std;class clsLoginScreen : protected clsScreen{private:	static void _Login()	{		_DrawScreenHeader("    Login Screen");		bool LoginFaild = false;		string UserName, Password;		do {			if (LoginFaild)			{				cout << "\nInvalaid UserName/Password!\n";			}			cout << "Enter UserName : ";			UserName = clsInputValidate::ReadString();			cout << "Enter Password : ";			Password = clsInputValidate::ReadString();			CurrentUser = clsUsers::Find(UserName, Password);			LoginFaild = CurrentUser.IsEmpty();		} while (LoginFaild);		clsMainMenue::ShowMainMenue();	}public:	static void ShowLoginScreen()	{		_Login();	}	};
