#include <iostream>#include "clsBankClient.h"void PrintClientRecordBalanceLine(clsBankClient Client){    cout << "|" << setw(15) << left << Client.AccountNumber();    cout << "|" << setw(30) << left << Client.FullName();    cout << "|" << setw(15) << left << Client.AccountBalance;}void ShowTotalBalances(){    vector<clsBankClient>vClients = clsBankClient::GetClientsList();    double TotalBalances = clsBankClient::CalculateTotalBalances();    cout << "\n\t\t|   Balances List (" << vClients.size() << ") Client.   |\n\t\t_______________________________\n\t ______________________________\n";    cout << "\t\t\t_______________________________";    cout << "\n________________________________________________________________________\n";    cout << "|" << left << setw(15)  << "Account Number";    cout << "|" << left << setw(30) << "Client Name";    cout << "|" << left << setw(15) << "Balance";    cout << "\n________________________________________________________________________\n";    if (vClients.size() == 0)    {        cout << "\t\t\t\tNo Clients Available.\n";    }    else    {    for (clsBankClient C : vClients)    {        PrintClientRecordBalanceLine(C);        cout << endl;    }    cout << "___________________________________________________________________________\n";    }    cout << "Total Balances : " << TotalBalances << " $.\n";    cout << "(" << clsUtil::NumberToText(TotalBalances) << "$).\n";}int main(){    ShowTotalBalances();    system("pause>0");    return 0;}
