#include <iostream>
using namespace std;

short ReadYear()
{

	short Year = 0;
	cout << "Enter a Year To Check : ";
	cin >> Year;
	return Year;

}

short ReadMonth()
{

	short Month = 0;

	do {
		cout << "Enter Month To Check : ";
		cin >> Month;
	} while (Month > 12 || Month < 1);

	return Month;
}

short ReadDay()
{

	short Day = 0;

	do {
		cout << "Enter Day To Check : ";
		cin >> Day;
	} while (Day > 31 || Day < 1);

	return Day;
}

struct sDate
{

	short Day;
	short Month;
	short Year;

};

sDate ReadFullDate()
{

	sDate Date;
	Date.Day = ReadDay();
	Date.Month = ReadMonth();
	Date.Year = ReadYear();
	cout << "\n\n";

	return Date;
}

bool IsDateOneLessThanDateTwo(sDate Date1, sDate Date2)
{
	return (Date1.Year > Date2.Year) ? true : ((Date1.Year == Date2.Year) ?
		(Date1.Month > Date2.Month ? true : (Date1.Month == Date2.Month ? Date1.Day > Date2.Day : false)) : false) ;
}

int main()
{

	sDate Date1 = ReadFullDate();
	sDate Date2 = ReadFullDate();


	 if (IsDateOneLessThanDateTwo(Date1, Date2))
		 cout << "\nYes, Date(1) is Less than Date(2). \n";
	 else
		 cout << "\nNo, Date(1) is NOT than Date(2). \n";



	return 0;
}
