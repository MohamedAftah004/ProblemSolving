bool IsLeapYear(short Year)
{

    return (Year % 400 == 0) || (Year % 100 != 0) && (Year % 4 == 0) ? true : false;

}
short NumberOfDaysInMonth(short Month, short Year)
{

    if (Month > 12 || Month < 1)
        return 0;

    short Days[12] = { 31 , 28 , 31 , 30 , 31 , 30 , 31 , 31 , 30 , 31 , 30 , 31 };

    return (Month == 2) ? (IsLeapYear(Year) ? 29 : 28) : Days[Month - 1];
}
bool IsLastDayInMonth(stDate Date)
{

    return (Date.Day == NumberOfDaysInMonth(Date.Month, Date.Year));

}
bool IsLastMonthInYear(short Month)
{
    return (Month == 12);
}
stDate IncreeseDateByOneDay(stDate Date)
{

    if (IsLastDayInMonth(Date))
    {
        if (IsLastMonthInYear(Date.Month))
        {
            Date.Month = 1;
            Date.Day = 1;
            Date.Year++;
        }
        else
        {
            Date.Day = 1;
            Date.Month++;
        }
    }
    else
    {
        Date.Day++;
    }

    return Date;
}
int GetDiffernceDaysInDate(stDate Date1, stDate Date2, bool IncludeEndDay = false)
{

    int Days = 0;
    while (IsDate1BeforeDate2(Date1, Date2))
    {

        Days++;
        Date1 = IncreeseDateByOneDay(Date1);
    }

    return IncludeEndDay ? ++Days : Days;
}


bool IsDate1BeforeDate2(stDate Date1, stDate Date2)
{
    return  (Date1.Year < Date2.Year) ? true :
        ((Date1.Year == Date2.Year) ? (Date1.Month < Date2.Month ? true :
            (Date1.Month == Date2.Month ? Date1.Day < Date2.Day : false)) : false);
}
bool IsDate1EqualsDate2(stDate Date1, stDate Date2)
{

    return (Date1.Day == Date2.Day && Date1.Month == Date2.Month && Date1.Year == Date2.Year) ? true : false;
}
bool IsDate1AfterDate2(stDate Date1, stDate Date2)
{

    return (!IsDate1BeforeDate2(Date1, Date2)) ? ((!IsDate1EqualsDate2(Date1, Date2)) ? true : false) : false;

}

enum enCompare { Before = -1, Equal = 0, After = 1 };
enCompare CompareDates(stDate Date1, stDate Date2)
{

    if (IsDate1BeforeDate2(Date1, Date2))
        return enCompare::Before;

    if (IsDate1EqualsDate2(Date1, Date2))
        return enCompare::Equal;


    return enCompare::After;

}

bool IsOverLapPeriod(stPeriod Period1, stPeriod Period2)
{

    if (
        CompareDates(Period2.DateEnd, Period1.DateStart) == enCompare::Before
        ||
        CompareDates(Period2.DateStart, Period1.DateEnd) == enCompare::After
        )
        return false;
    else
        return true;

}

bool IsDateInPeriod(stDate Date, stPeriod Period)
{

    // before -> less than   -   after -> greatest than

    return !(CompareDates(Date, Period.DateStart) == enCompare::Before || CompareDates(Date, Period.DateEnd) == enCompare::After);

}


short OverLapCount(stPeriod Period1, stPeriod Period2)
{

    return  GetDiffernceDaysInDate(Period2.DateStart, Period1.DateEnd);

}

int PeriodLengthInDays(stPeriod Period, bool IncludeEndDate = false)
{
    return GetDiffernceDaysInDate(Period.DateStart, Period.DateEnd, IncludeEndDate);
}



int CountOVerLapDays(stPeriod Period1, stPeriod Period2)
{

    int Period1Length = PeriodLengthInDays(Period1, true);
    int Period2Length = PeriodLengthInDays(Period2, true);
    int OverLapDays = 0;

    if (!IsOverLapPeriod(Period1, Period2))
    {
        return 0;
    }

    if (Period1Length < Period2Length)
    {

        while (IsDate1BeforeDate2(Period1.DateStart, Period2.DateEnd))
        {


            if (IsDateInPeriod(Period1.DateStart, Period2))
                OverLapDays++;


            Period1.DateStart = IncreeseDateByOneDay(Period1.DateStart);

        }
    }
    else
    {

        while (IsDate1BeforeDate2(Period2.DateStart, Period2.DateEnd))
        {

            if (IsDateInPeriod(Period2.DateStart, Period1))
                OverLapDays++;

            Period2.DateStart = IncreeseDateByOneDay(Period2.DateStart);

        }

    }


    return OverLapDays;

}


bool IsValidDate(stDate Date)
{


    if (Date.Day > 31 || Date.Day < 1)
        return false;

    if (Date.Month > 12 || Date.Month < 1)
        return false;



    if (Date.Month == 2)
    {
        if (IsLeapYear(Date.Year))

        {
            if (Date.Day > 29)
                return false;
        }
        else
        {

            if (Date.Day > 28)
                return false;
        }

    }
    
    short DaysInMonth = NumberOfDaysInMonth(Date.Month, Date.Year);
    if (Date.Day > DaysInMonth)
        return false;
    
    return true;



}
