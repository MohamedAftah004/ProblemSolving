
#include <iostream>
using namespace std;


struct stDate {

	short Year;
	short Month;
	short Day;

};

short ReadYear()
{

	short Year = 0;
	cout << "Enter a Year To Check : ";
	cin >> Year;
	return Year;

}
short ReadMonth()
{

	short Month = 0;

	do {
		cout << "Enter Month To Check : ";
		cin >> Month;
	} while (Month > 12 || Month < 1);

	return Month;
}
short ReadDay()
{

	short Day = 0;

	do {
		cout << "Enter Day To Check : ";
		cin >> Day;
	} while (Day > 31 || Day < 1);

	return Day;
}

stDate ReadFullDate()
{
    stDate Date ;
    Date.Day = ReadDay() ;
    Date.Month = ReadMonth() ;
    Date.Year = ReadYear() ;

    return Date;
}

bool IsLeapYear(short Year)
{

	return (Year % 400 == 0) || (Year % 100 != 0) && (Year % 4 == 0) ? true : false;

}
short NumberOfDaysInMonth(short Month , short Year)
{

    if(Month > 12 || Month <1)
    return 0;

	short Days[12] = { 31 , 28 , 31 , 30 , 31 , 30 , 31 , 31 , 30 , 31 , 30 , 31 };

	return (Month == 2) ? (IsLeapYear(Year) ? 29 : 28) : Days[Month - 1];
}

bool IsLastDayInMonth(stDate Date)
{

    return (Date.Day == NumberOfDaysInMonth(Date.Month, Date.Year));

}
bool IsLastMonthInYear(short Month)
{
    return (Month == 12);
}

// 1 day
stDate IncreaseDateByOneDay(stDate Date)
{

    if(IsLastDayInMonth(Date))
    {
        if(IsLastMonthInYear(Date.Month))
        {
            Date.Month = 1 ;
            Date.Day = 1 ;
            Date.Year++;
        }
        else
        {
            Date.Day = 1;
            Date.Month++;
        }
    }
    else
    {
        Date.Day++;
    }

    return Date;
}
// x days
stDate IncreaseDateByMoreDays(stDate Date , short MoreDays)
{

    for(int i = 1 ; i <= MoreDays ; i++)
    {
        Date = IncreaseDateByOneDay(Date);
    }

    return Date;
}

// 1 week
stDate IncreaseDateByOneWeek(stDate Date)
{

    for(int i = 1 ; i<= 7 ;i++)
    {
        Date = IncreaseDateByOneDay(Date);
    }

    return Date;
}
// x weeks
stDate IncreaseDateByMoreWeeks(stDate Date , short MoreWeeks)
{

    for(int i = 1 ; i<= MoreWeeks ;i++)
    {
        Date = IncreaseDateByOneWeek(Date);
    }

    return Date;
}

// 1 month
stDate IncreaseDateByOneMonth(stDate Date )
{

    if(Date.Month == 12)
    {
        Date.Month = 1;
        Date.Year++;
    }
else
    {
    Date.Month++;
    }


    short NumberOfDateInCurrentMonth = NumberOfDaysInMonth(Date.Month , Date.Year);
    if(Date.Day > NumberOfDateInCurrentMonth)
    {
        Date.Day = NumberOfDateInCurrentMonth ;
    }

    return Date;
}

// x months
stDate IncreaseDateByMoreMonths(stDate Date , short MoreMonths)
{

    for(int i = 1 ; i<= MoreMonths ;i++)
    {
        Date = IncreaseDateByOneMonth(Date);
    }

    return Date;
}


// 1 year
stDate IncreaseDateByOneYear(stDate Date)
{

    Date.Year++;

    return Date;
}
// 10 years
stDate IncreaseDateByMoreYears(stDate Date , short MoreYears)
{

    for(int i = 1 ; i<= MoreYears ; i++)
    {
            Date = IncreaseDateByOneYear(Date);
    }

    return Date;
}
// 10 years faster
stDate IncreaseDateByMoreYearsFaster(stDate Date , short MoreYears) {

     Date.Year += MoreYears;
     return Date;

}


// 1 Decade
stDate IncreaseDateByOneDecade(stDate Date)
{
    Date.Year += 10;
    return Date;
}
// 10 Decade
stDate IncreaseDateByMoreDecade(stDate Date , short MoreDecade)
{

    for(int i = 1 ; i<= MoreDecade*10 ; i++)
    {
            Date = IncreaseDateByOneDecade(Date);
    }

    return Date;
}
// 10 Decade faster
stDate IncreaseDateByMoreDecadeFaster(stDate Date , short MoreDecade)
{

    Date.Year += MoreDecade*10 ;

    return Date;
}


// 1 Century
stDate IncreaseDateByOneCentury(stDate Date)
{
    Date.Year += 100;
    return Date;
}

// 1 Millennium
stDate IncreaseDateByOneMillennium(stDate Date)
{
    Date.Year += 1000;
    return Date;
}
