#include <iostream>
#include <string>
using namespace std;


short ReadYear()
{
	cout << "Enter Year : ";
	int Year = 0; cin >> Year; return Year;
}

bool IsLeapYear(short Year)
{

	if (Year % 400 == 0)
	{
		return true;
	}

	else if (Year % 100 == 0)
	{
		return false;
	}

	else if (Year % 4 == 0)
	{
		return true;
	}

	else
		return false;

}


int main()
{

	short Year = ReadYear();

	if (IsLeapYear(Year) == true)
		cout << "\nYes , Year [" << Year << "] Is a leap year.\n" ;
	else
		cout << "\nNo , Year [" << Year << "] Is a NOT leap year.\n";

	system("pause>0");
	return 0;
}
